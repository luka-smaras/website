using System;
using System.Collections.Generic;
using System.ComponentModel;
using System.Data;
using System.Drawing;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
using System.Windows.Forms;
using System.IO;

namespace Kviz
{
    public partial class Form1 : Form
    {
        public int odgk, odgk1, odgk2, odgk3, odgk4;
        int p1, p2, p3, p4, p5;
        int ukupnoPitanja = 10; // broj pitanja u mini kvizu
        int trenutnoPitanje = 0;

        public Form1()
        {
            InitializeComponent();
            tabControl1.SelectedTab = tabPage1;
            tabControl1.Appearance = TabAppearance.FlatButtons;
            tabControl1.ItemSize = new Size(0, 1);
            tabControl1.SizeMode = TabSizeMode.Fixed;
        }

        private void button1_Click(object sender, EventArgs e)
        {
            tabControl1.SelectTab(1);
            timer1.Start();
            listBox1.Items.Clear();
            listBox1.Items.AddRange(File.ReadAllLines(@"pitanja.txt"));

            // Postavimo progress bar
            progressBar1.Maximum = ukupnoPitanja;
            progressBar1.Value = 0;
            trenutnoPitanje = 0;

            // Prikaži prvo pitanje
            button2.PerformClick();
        }

        int time = 0;
        int poeni = 0;
        int preskoceno = 0;

        private void timer1_Tick(object sender, EventArgs e)
        {
            time++;
            label5.Text = "Vreme:" + time;
        }

        private void checkBox1_CheckedChanged(object sender, EventArgs e)
        {
            p1 = checkBox1.Checked ? 1 : 0;
        }
        private void checkBox2_CheckedChanged(object sender, EventArgs e)
        {
            p2 = checkBox2.Checked ? 1 : 0;
        }
        private void checkBox3_CheckedChanged(object sender, EventArgs e)
        {
            p3 = checkBox3.Checked ? 1 : 0;
        }
        private void checkBox4_CheckedChanged(object sender, EventArgs e)
        {
            p4 = checkBox4.Checked ? 1 : 0;
        }
        private void checkBox5_CheckedChanged(object sender, EventArgs e)
        {
            p5 = checkBox5.Checked ? 1 : 0;
        }

        private void button2_Click(object sender, EventArgs e)
        {
            // Random pitanje
            int totalQuestions = listBox1.Items.Count / 11;
            Random rnd = new Random();
            int p = rnd.Next(0, totalQuestions) * 11;

            label1.Text = listBox1.Items[p].ToString();
            checkBox1.Text = listBox1.Items[p + 1].ToString();
            checkBox2.Text = listBox1.Items[p + 2].ToString();
            checkBox3.Text = listBox1.Items[p + 3].ToString();
            checkBox4.Text = listBox1.Items[p + 4].ToString();
            checkBox5.Text = listBox1.Items[p + 5].ToString();

            odgk = Convert.ToInt16(listBox1.Items[p + 6]);
            odgk1 = Convert.ToInt16(listBox1.Items[p + 7]);
            odgk2 = Convert.ToInt16(listBox1.Items[p + 8]);
            odgk3 = Convert.ToInt16(listBox1.Items[p + 9]);
            odgk4 = Convert.ToInt16(listBox1.Items[p + 10]);

            // reset checkboxovi
            checkBox1.Checked = false;
            checkBox2.Checked = false;
            checkBox3.Checked = false;
            checkBox4.Checked = false;
            checkBox5.Checked = false;
        }

        private void button3_Click(object sender, EventArgs e)
        {
            int tacni = 0;
            bool greska = false;

            if (p1 == 1 && odgk == 1) tacni++;
            if (p1 == 0 && odgk == 1) greska = true;

            if (p2 == 1 && odgk1 == 1) tacni++;
            if (p2 == 0 && odgk1 == 1) greska = true;

            if (p3 == 1 && odgk2 == 1) tacni++;
            if (p3 == 0 && odgk2 == 1) greska = true;

            if (p4 == 1 && odgk3 == 1) tacni++;
            if (p4 == 0 && odgk3 == 1) greska = true;

            if (p5 == 1 && odgk4 == 1) tacni++;
            if (p5 == 0 && odgk4 == 1) greska = true;

            int osvojeniPoeni = greska ? -10 : tacni * 10;
            poeni += osvojeniPoeni;
            label12.Text = "Poeni: " + poeni;

            // Pomeranje progress bara
            trenutnoPitanje++;
            progressBar1.Value = trenutnoPitanje;

            if (trenutnoPitanje >= ukupnoPitanja)
            {
                MessageBox.Show("Kraj kviza sa osvojenih poena: " + poeni);
                Reset();
                timer1.Stop();
                return;
            }
            else
            {
                button2.PerformClick(); // sledeće pitanje
            }
        }

        private void button4_Click(object sender, EventArgs e)
        {
            preskoceno++;
            label6.Text = "Broj preskocenih pitanja:" + preskoceno;

            trenutnoPitanje++;
            progressBar1.Value = trenutnoPitanje;

            if (trenutnoPitanje >= ukupnoPitanja)
            {
                MessageBox.Show("Kraj kviza sa osvojenih poena:" + poeni);
                Reset();
                timer1.Stop();
                return;
            }
            else
            {
                button2.PerformClick(); // sledeće pitanje
            }
        }

        private void Reset()
        {
            timer1.Start();
            poeni = 0;
            label12.Text = "Poeni:" + poeni;
            preskoceno = 0;
            label6.Text = "Broj preskocenih pitanja:" + preskoceno;
            time = 0;
            label5.Text = "Vreme:" + time;
            progressBar1.Value = 0;
            trenutnoPitanje = 0;
        }
    }
